تصدير البيانات لـ Excel


// مصفوفة تخزين العمليات محلياً في المتصفح let db = JSON.parse(localStorage.getItem('masar_db')) || []; // تشغيل العدادات والجدول فور تحميل الصفحة document.addEventListener("DOMContentLoaded", () => { doRender(); doCount(); }); // التبديل السريع بين تبويبات التطبيق function goTab(t) { document.querySelectorAll('.sec').forEach(s => s.classList.remove('active')); document.querySelectorAll('.nav-i').forEach(b => b.classList.remove('active')); document.getElementById(`sec-${t}`).classList.add('active'); document.getElementById(`t-${t}`).classList.add('active'); } // حفظ حركة استلام أو تسليم جديدة function doSave(e) { e.preventDefault(); db.unshift({ type: document.getElementById('i-type').value, plate: document.getElementById('i-plate').value, vin: document.getElementById('i-vin').value || '-', owner: document.getElementById('i-owner').value, location: document.getElementById('i-loc').value, status: document.getElementById('i-status').value, carrier: document.getElementById('i-carrier').value || '-', cplate: document.getElementById('i-cplate').value || '-', driver: document.getElementById('i-driver').value || '-', dest: document.getElementById('i-dest').value || '-' }); localStorage.setItem('masar_db', JSON.stringify(db)); document.getElementById('f-car').reset(); doRender(); doCount(); goTab('home'); } // تحديث لوحة عدادات حصر السيارات الفورية function doCount() { document.getElementById('s-total').innerText = db.length; document.getElementById('s-yard').innerText = db.filter(c => c.location === 'الساحة').length; document.getElementById('s-main').innerText = db.filter(c => c.location === 'الصيانة').length; document.getElementById('s-road').innerText = db.filter(c => c.location === 'في الطريق').length; } // بناء وعرض جدول حركات السيارات function doRender(items = db) { const tbody = document.getElementById('table-body'); tbody.innerHTML = ''; items.forEach(c => { const tr = document.createElement('tr'); tr.innerHTML = ` ${c.type} ${c.plate} ${c.owner} ${c.location} ${c.carrier} ${c.cplate} ${c.driver} ${c.dest} ${c.status} `; tbody.appendChild(tr); }); } // محرك البحث والفلترة المتقدمة متعدد الخيارات (الوجهة، الناقل، السائق، اللوحة) function runFilter() { const g = document.getElementById('q-gen').value.toLowerCase(); const c = document.getElementById('q-car').value.toLowerCase(); const d = document.getElementById('q-drv').value.toLowerCase(); const s = document.getElementById('q-dst').value.toLowerCase(); const l = document.getElementById('q-loc').value; const res = db.filter(i => { return (i.plate.toLowerCase().includes(g) || i.owner.toLowerCase().includes(g)) && (i.carrier.toLowerCase().includes(c) || i.cplate.toLowerCase().includes(c)) && i.driver.toLowerCase().includes(d) && i.dest.toLowerCase().includes(s) && (l === "" || i.location === l); }); doRender(res); } // تصدير كشوفات حركة المركبات إلى Excel (CSV) function toCSV() { if(!db.length) return alert('لا توجد بيانات لتصديرها حالياً!'); let csv = "\ufeffنوع العملية,رقم اللوحة,الهيكل,المالك,الموقع,الحالة,الناقل,لوحة النقل,السائق,الوجهة\n"; db.forEach(c => { csv += `"${c.type}","${c.plate}","${c.vin}","${c.owner}","${c.location}","${c.status}","${c.carrier}","${c.cplate}","${c.driver}","${c.dest}"\n`; }); const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' }); const a = document.createElement("a"); a.href = URL.createObjectURL(blob); a.download = `تقرير_مسار_السيارات.csv`; a.click(); } // إنشاء نسخة احتياطية لقاعدة البيانات المحلية function toBackup() { const a = document.createElement('a'); a.href = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(db)); a.download = `نسخة_مسار.json`; a.click(); }
